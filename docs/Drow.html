Example
// <my-counter></my-counter>

const config = {
  name: "my-counter",
  state: { count: 0 },
  css: `button { color: red; }`,
  template: `<button @click="increment">Count is {{count}}</button>`,
  methods: {
    increment() {
      this.state.count++;
    }
  }
}

Drow.register(config);
